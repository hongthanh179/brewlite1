const rand = () => {
  state = (state * 1664525 + 1013904223) % 4294967296;
  return state / 4294967296;

const randInt = (min: number, max: number) =>
  min + Math.floor(rand() * (max - min + 1));
const pick = <T>(arr: T[]): T => arr[Math.floor(rand() * arr.length)];
const shuffle = <T>(arr: T[]): T[] => {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
};

function calcDiscount(coupon: Coupon, subtotal: number): number {
  const raw =
    coupon.type === DiscountType.PERCENT
      ? Math.floor((subtotal * coupon.value) / 100)
      : coupon.value;
  const capped = coupon.maxDiscount ? Math.min(raw, coupon.maxDiscount) : raw;
  return Math.min(capped, subtotal);
}

import { NestFactory } from '@nestjs/core';
import { NestExpressApplication } from '@nestjs/platform-express';
import { join } from 'path';
import { existsSync, mkdirSync } from 'fs';
import { AppModule } from './app.module';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { ValidationPipe } from '@nestjs/common';
import { TransformInterceptor } from './common/interceptors/transform.interceptor';
import { HttpExceptionFilter } from './common/filters/http-exception.filter';

async function bootstrap() {
  const app = await NestFactory.create<NestExpressApplication>(AppModule);

  // 1. Thêm tiền tố toàn cục /api cho toàn bộ endpoint
  app.setGlobalPrefix('api');

  // 2. Đảm bảo thư mục uploads luôn tồn tại khi chạy cục bộ để không bị crash
  const uploadDir = join(process.cwd(), 'uploads');
  if (!existsSync(uploadDir)) {
    mkdirSync(uploadDir, { recursive: true });
  }

  // Mở quyền truy cập thư mục uploads qua đường dẫn /uploads/...
  app.useStaticAssets(uploadDir, {
    prefix: '/uploads/',
  });

  // 3. Mở CORS động: Tự động cho phép mọi client (localhost, Render frontend, preview domains)
  // Kết hợp cùng credentials: true mà không sợ lỗi chặn header Access-Control-Allow-Origin
  app.enableCors({
    origin: true,
    methods: 'GET,HEAD,PUT,PATCH,POST,DELETE,OPTIONS',
    credentials: true,
  });

  // 4. Thiết lập Validation Pipe toàn cục
  app.useGlobalPipes(
      new ValidationPipe({
        whitelist: true,
        stopAtFirstError: true,
        transform: true,
        transformOptions: { enableImplicitConversion: true },
      }),
  );

  // 5. Cấu hình Swagger API Docs
  const config = new DocumentBuilder()
      .setTitle('BrewLite API')
      .setDescription('API cho hệ thống đặt nước BrewLite')
      .setVersion('1.0')
      .addBearerAuth()
      .build();

  const document = SwaggerModule.createDocument(app, config);
  SwaggerModule.setup('api/docs', app, document);

  app.useGlobalInterceptors(new TransformInterceptor());
  app.useGlobalFilters(new HttpExceptionFilter());

  // 6. Lắng nghe cổng và tự động nhận diện Domain hiển thị log
  const port = process.env.PORT || 3001;
  await app.listen(port, '0.0.0.0');

  const baseUrl = process.env.RENDER_EXTERNAL_URL || `http://localhost:${port}`;

  console.log(`[NestJS] API Server running on: ${baseUrl}/api`);
  console.log(`[NestJS] Swagger Docs running on: ${baseUrl}/api/docs`);
}
bootstrap();

import {
  ExceptionFilter,
  Catch,
  ArgumentsHost,
  HttpException,
} from '@nestjs/common';
import { Response, Request } from 'express';

@Catch(HttpException)
export class HttpExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();
    const status = exception.getStatus();
    const exceptionResponse: any = exception.getResponse();

    response.status(status).json({
      statusCode: status,
      // Ưu tiên lấy mảng message do ValidationPipe (DTO) trả ra, nếu không có thì lấy message mặc định
      message: exceptionResponse.message || exception.message,
      error: exceptionResponse.error || 'Lỗi hệ thống',
      path: request.url,
      timestamp: new Date().toISOString(),
    });
  }
}

import {
  Injectable,
  NestInterceptor,
  ExecutionContext,
  CallHandler,
} from '@nestjs/common';
import { Observable } from 'rxjs';
import { map } from 'rxjs/operators';

@Injectable()
export class TransformInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const ctx = context.switchToHttp();
    const response = ctx.getResponse();

    return next.handle().pipe(
      map((data) => ({
        statusCode: response.statusCode,
        message: data?.message || 'Thao tác thành công',
        // Tránh bọc data 2 lần nếu trong Service bạn đã trả về object có key data
        data: data?.data !== undefined ? data.data : data,
      })),
    );
  }
}

    } else {
      cb(
        new BadRequestException(
          `Định dạng file không hợp lệ. Chỉ chấp nhận các file ảnh (jpg, jpeg, png, gif, webp).`,
        ),
        false,
      );
    }
  },

  // 3. Giới hạn dung lượng (Chống DDoS đầy ổ cứng)
  limits: {
    fileSize: 5 * 1024 * 1024, // Tối đa 5MB / 1 file
  },
};

import {
  Injectable,
  UnauthorizedException,
  ForbiddenException,
  BadRequestException,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';
import { UsersService } from '../users/users.service';
import { CreateUserDto } from '../users/dto/create-user.dto';
import { ChangePasswordDto } from '../users/dto/change-password.dto';import { Role } from '@prisma/client';

@Injectable()
export class AuthService {
  constructor(
      private prisma: PrismaService,
      private jwtService: JwtService,
      private usersService: UsersService,
  ) {}

  async register(createUserDto: CreateUserDto) {
    return this.usersService.create(createUserDto, Role.CUSTOMER);
  }

  async login(email: string, pass: string) {
    const user = await this.prisma.user.findUnique({ where: { email } });
    if (!user) {
      throw new UnauthorizedException(
          'Tài khoản hoặc mật khẩu không chính xác',
      );
    }

    if (!user.isActive) {
      throw new ForbiddenException(
          'Tài khoản của bạn đã bị khóa. Vui lòng liên hệ Quản trị viên để biết thêm chi tiết.',
      );
    }

    const isPasswordMatching = await bcrypt.compare(pass, user.passwordHash);
    if (!isPasswordMatching) {
      throw new UnauthorizedException(
          'Tài khoản hoặc mật khẩu không chính xác',
      );
    }

    const payload = { sub: user.id, email: user.email, role: user.role };

    return {
      access_token: await this.jwtService.signAsync(payload),
      user: {
        id: user.id,
        name: user.name,
        role: user.role,
        loyaltyPoints: user.loyaltyPoints,
      },
    };
  }

  // ========================================================
  // ENDPOINT ĐỔI MẬT KHẨU (Khớp với ChangePasswordDto & passwordHash)
  // ========================================================
  async changePassword(userId: number, dto: ChangePasswordDto) {
    if (dto.newPassword !== dto.confirmNewPassword) {
      throw new BadRequestException('Xác nhận mật khẩu mới không khớp');
    }

    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (!user) {
      throw new NotFoundException('Không tìm thấy thông tin tài khoản');
    }

    // 1. So khớp mật khẩu cũ với cột passwordHash trong DB
    const isOldPasswordMatch = await bcrypt.compare(
        dto.oldPassword,
        user.passwordHash,
    );
    if (!isOldPasswordMatch) {
      throw new BadRequestException('Mật khẩu cũ không chính xác');
    }

    // 2. Không cho phép đặt trùng mật khẩu cũ
    const isSameAsOld = await bcrypt.compare(
        dto.newPassword,
        user.passwordHash,
    );
    if (isSameAsOld) {
      throw new BadRequestException(
          'Mật khẩu mới không được trùng với mật khẩu hiện tại',
      );
    }

    // 3. Băm mật khẩu mới và cập nhật vào cột passwordHash
    const newPasswordHash = await bcrypt.hash(dto.newPassword, 10);
    await this.prisma.user.update({
      where: { id: userId },
      data: { passwordHash: newPasswordHash },
    });

    return {
      success: true,
      message: 'Đổi mật khẩu thành công',
    };
  }
}

import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { Role } from '@prisma/client';
import { ROLES_KEY } from '../decorators/roles.decorator';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<Role[]>(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);

    // Nếu API không yêu cầu quyền gì, cho phép qua
    if (!requiredRoles) {
      return true;
    }

    const { user } = context.switchToHttp().getRequest();

    // Kiểm tra quyền
    if (!user || !user.role) {
      throw new ForbiddenException('Không tìm thấy thông tin định danh');
    }

    const hasRole = requiredRoles.includes(user.role);
    if (!hasRole) {
      throw new ForbiddenException(
        'Chỉ Quản trị viên (Admin) mới có quyền thực hiện hành động này',
      );
    }
    return true;
  }
}

import {

  BadRequestException,
} from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateUserDto } from './dto/create-user.dto';
import { UpdateUserDto } from './dto/update-user.dto';

import * as bcrypt from 'bcrypt';

@Injectable()
export class UsersService {
  constructor(private prisma: PrismaService) {}

  async create(createUserDto: CreateUserDto, role: Role = Role.CUSTOMER) {
    // 1. Kiểm tra trùng Email
    const existingEmail = await this.prisma.user.findUnique({
      where: { email: createUserDto.email },
    });
    if (existingEmail) {
      throw new ConflictException('Email này đã được sử dụng');
    }

    // 2. Kiểm tra trùng Số điện thoại
    if (createUserDto.phone) {
      const existingPhone = await this.prisma.user.findUnique({
        where: { phone: createUserDto.phone },
      });
      if (existingPhone) {
        throw new ConflictException('Số điện thoại này đã được sử dụng');
      }
    }

    const saltRounds = 10;
    const hashedPassword = await bcrypt.hash(
      createUserDto.password,
      saltRounds,
    );

    const newUser = await this.prisma.user.create({
      data: {
        email: createUserDto.email,
        name: createUserDto.name,
        phone: createUserDto.phone,
        gender: createUserDto.gender,
        passwordHash: hashedPassword,
        role: role,
      },
      select: {
        id: true,
        email: true,
        name: true,
        phone: true,
        gender: true,
        loyaltyPoints: true,
        role: true,
      },
    });

    return { message: 'Tạo tài khoản thành công', data: newUser };
  }

  // Cập nhật hàm findAll để hỗ trợ phân trang và tìm kiếm
  async findAll(query: QueryUserDto) {
    const { search, role, page, limit } = query;
    const skip = (page - 1) * limit;

    const whereCondition: any = {};

    if (role) {
      whereCondition.role = role;
    }

    if (search) {
      // Tìm kiếm tương đối trên name, email và phone
      whereCondition.OR = [
        { name: { contains: search } },
        { email: { contains: search } },
        { phone: { contains: search } },
      ];
    }

    const [data, total] = await Promise.all([
      this.prisma.user.findMany({
        where: whereCondition,
        select: {
          id: true,
          email: true,
          name: true,
          phone: true,
          gender: true,
          loyaltyPoints: true,
          role: true,
          isActive: true,
        },
        skip,
        take: limit,
        orderBy: { id: 'desc' },
      }),
      this.prisma.user.count({ where: whereCondition }),
    ]);

    return {
      data,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async findOne(id: number) {
    const user = await this.prisma.user.findUnique({
      where: { id },
      select: {
        id: true,
        email: true,
        name: true,
        phone: true,
        gender: true,

  async changePassword(userId: number, changePasswordDto: ChangePasswordDto) {
    if (
      changePasswordDto.newPassword !== changePasswordDto.confirmNewPassword
    ) {
      throw new BadRequestException('Xác nhận mật khẩu không khớp!');
    }

    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (!user) throw new NotFoundException('Người dùng không tồn tại');

    const isMatch = await bcrypt.compare(

    if (!isMatch) throw new BadRequestException('Mật khẩu cũ không chính xác');

    const saltRounds = 10;

      changePasswordDto.newPassword,
      saltRounds,
    );

    await this.prisma.user.update({
      where: { id: userId },
      data: { passwordHash: hashedNewPassword },
    });

    return { message: 'Đổi mật khẩu thành công. Vui lòng đăng nhập lại.' };
  }

  async update(id: number, updateUserDto: UpdateUserDto) {
    const user = await this.prisma.user.findUnique({ where: { id } });
    if (!user)
      throw new NotFoundException(`Không tìm thấy người dùng với ID ${id}`);

    if (updateUserDto.phone && updateUserDto.phone !== user.phone) {
      const existingPhone = await this.prisma.user.findFirst({
        where: {
          phone: updateUserDto.phone,
          id: { not: id },
        },
      });
      if (existingPhone) {
        throw new ConflictException(
          'Số điện thoại này đã được sử dụng bởi một tài khoản khác',
        );
      }
    }

    // --- BẮT ĐẦU ĐOẠN SỬA LỖI BĂM MẬT KHẨU ---
    const updateData: any = { ...updateUserDto };

    // Nếu Admin điền mật khẩu mới ở giao diện Edit
    if (updateData.password) {
      // 1. Băm mật khẩu ra
      const saltRounds = 10;
      updateData.passwordHash = await bcrypt.hash(
        updateData.password,
        saltRounds,
      );

      // 2. Xóa cái trường password chữ thường đi để tránh Prisma báo lỗi
      delete updateData.password;
    }
    // --- KẾT THÚC ĐOẠN SỬA LỖI ---

    return this.prisma.user.update({
      where: { id },
      data: updateData, // Dùng updateData thay vì updateUserDto
      select: {
        id: true,
        email: true,
        name: true,
        phone: true,
        gender: true,
        loyaltyPoints: true,
        role: true,
        isActive: true, // Nhớ thêm isActive để trả về Frontend
      },
    });
  }

  async toggleStatus(id: number) {

    if (!user)
      throw new NotFoundException(`Không tìm thấy người dùng với ID ${id}`);

    //if (user.role === Role.ADMIN) {
    //throw new ConflictException('Không thể khóa tài khoản Admin!');
    //}

    const updatedUser = await this.prisma.user.update({
      where: { id },
      data: { isActive: !user.isActive },
      select: { id: true, name: true, isActive: true },
    });

    return {
      message: updatedUser.isActive
        ? `Đã mở khóa tài khoản ${updatedUser.name}`
        : `Đã khóa tài khoản ${updatedUser.name}`,
    };
  }
}

import { ApiProperty } from '@nestjs/swagger';
import {
  IsEmail,
  IsNotEmpty,
  IsOptional,
  IsString,
  Matches,
  MaxLength,
  MinLength,
  IsIn,
} from 'class-validator';

export class CreateUserDto {
  @ApiProperty({
    example: 'khachhang@gmail.com',
    description: 'Email đăng nhập',
  })
  @MaxLength(255, { message: 'Email không được vượt quá 255 ký tự' })
  @IsEmail({}, { message: 'Email không đúng định dạng' })
  @IsNotEmpty({ message: 'Email không được để trống' })
  email: string;

  @ApiProperty({
    example: 'Abc@1234',
    description: 'Mật khẩu bảo mật cao',
  })
  @Matches(
    /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$/,
    {
      message:
        'Mật khẩu phải chứa ít nhất 1 chữ hoa, 1 chữ thường, 1 chữ số và 1 ký tự đặc biệt',

  )
  @MaxLength(50, { message: 'Mật khẩu không được vượt quá 50 ký tự' })
  @MinLength(8, { message: 'Mật khẩu phải từ 8 ký tự trở lên' })

    description: 'Họ và tên',
    required: false,

  })
  @IsString({ message: 'Họ và tên phải là chuỗi ký tự' })
  @IsOptional()
  name?: string;

  @ApiProperty({
    example: '0901234567',
    required: false,
    description: 'Số điện thoại VN',
  })

import { Transform, Type } from 'class-transformer';
import { IsInt, IsOptional, IsString, Max, Min, IsEnum } from 'class-validator';
import { Role } from '@prisma/client';

export class QueryUserDto {
  @ApiPropertyOptional({
    description: 'Tìm theo tên, email hoặc số điện thoại',
  })
  @Transform(({ value }) => (typeof value === 'string' ? value.trim() : value))
  @IsOptional()
  @IsString()
  search?: string;

  @ApiPropertyOptional({
    enum: Role,
    description: 'Lọc theo quyền (ADMIN / CUSTOMER)',

  @IsOptional()
  @IsEnum(Role)
  role?: Role;

  @ApiPropertyOptional({ type: Number, default: 1 })

  @IsInt()
  @Min(1)
  page: number = 1;

  @ApiPropertyOptional({ type: Number, default: 20, maximum: 100 })
  @Type(() => Number)
  @IsOptional()
  @IsInt()
  @Min(1)
  @Max(100)
  limit: number = 20;
}
